cmake_minimum_required(VERSION 3.16)
project(partymimic_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(Threads REQUIRED)

# ---- Eigen: reuse the robot's system copy (same as wbc_nx/remote_start); optional bundled fallback third_party/eigen3
find_path(EIGEN3_INCLUDE_DIR Eigen/Dense HINTS /usr/include/eigen3 /usr/local/include/eigen3 ${CMAKE_SOURCE_DIR}/third_party/eigen3)
if(NOT EIGEN3_INCLUDE_DIR)
  message(FATAL_ERROR "Eigen3 headers not found (expected /usr/include/eigen3 as used by wbc_nx/remote_start)")
endif()

# ---- ONNX Runtime: PM_ORT_ROOT (default: bundled aarch64 build in third_party/onnxruntime; build.sh prefers a system copy when present)
set(PM_ORT_ROOT "${CMAKE_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "ONNX Runtime root (include/ + lib/)")
find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h HINTS ${PM_ORT_ROOT}/include PATH_SUFFIXES onnxruntime NO_DEFAULT_PATH)
find_library(ORT_LIB NAMES onnxruntime HINTS ${PM_ORT_ROOT}/lib ${PM_ORT_ROOT}/lib/aarch64-linux-gnu NO_DEFAULT_PATH)
if(NOT ORT_INCLUDE_DIR OR NOT ORT_LIB)
  message(FATAL_ERROR "ONNX Runtime not found under ${PM_ORT_ROOT} (need include/onnxruntime_cxx_api.h and lib/libonnxruntime.so)")
endif()
get_filename_component(ORT_LIB_DIR ${ORT_LIB} DIRECTORY)

# ---- Unitree SDK2 (+ CycloneDDS) as installed on the robot in /usr/local (see wbc_nx/remote_start/CMakeLists.txt)
set(SDK_HINT_LIB /usr/local/lib $ENV{UNITREE_SDK2_DIR}/lib/aarch64 $ENV{UNITREE_SDK2_DIR}/lib/x86_64 /opt/unitree_robotics/lib $ENV{HOME}/unitree_sdk2/lib/aarch64)
find_library(SDK2_LIB unitree_sdk2 HINTS ${SDK_HINT_LIB})
find_library(DDSC_LIB ddsc HINTS ${SDK_HINT_LIB} $ENV{UNITREE_SDK2_DIR}/thirdparty/lib/aarch64 $ENV{UNITREE_SDK2_DIR}/thirdparty/lib/x86_64)
find_library(DDSCXX_LIB ddscxx HINTS ${SDK_HINT_LIB} $ENV{UNITREE_SDK2_DIR}/thirdparty/lib/aarch64 $ENV{UNITREE_SDK2_DIR}/thirdparty/lib/x86_64)
find_path(SDK2_INCLUDE_DIR unitree/robot/channel/channel_factory.hpp HINTS /usr/local/include /opt/unitree_robotics/include $ENV{UNITREE_SDK2_DIR}/include $ENV{HOME}/unitree_sdk2/include)
find_path(DDSCXX_INCLUDE_DIR dds/dds.hpp HINTS /usr/local/include/ddscxx $ENV{UNITREE_SDK2_DIR}/thirdparty/include/ddscxx)
find_path(DDSC_INCLUDE_DIR dds/dds.h HINTS /usr/local/include/ddsc /usr/local/include $ENV{UNITREE_SDK2_DIR}/thirdparty/include/ddsc)
find_path(ICEORYX_INCLUDE_DIR iceoryx_hoofs HINTS /usr/local/include/iceoryx/v2.0.2 $ENV{UNITREE_SDK2_DIR}/thirdparty/include/iceoryx/v2.0.2)
foreach(v SDK2_LIB DDSC_LIB DDSCXX_LIB SDK2_INCLUDE_DIR DDSCXX_INCLUDE_DIR)
  if(NOT ${v})
    message(FATAL_ERROR "${v} not found: Unitree SDK2 (libunitree_sdk2, libddsc, libddscxx) must be installed in /usr/local like for wbc_nx/remote_start")
  endif()
endforeach()

set(CMAKE_BUILD_RPATH "${ORT_LIB_DIR};/usr/local/lib")
set(CMAKE_INSTALL_RPATH "${ORT_LIB_DIR};/usr/local/lib")

# ---- policy core: Eigen + ONNX Runtime only (no robot SDK)
add_library(pm_policy STATIC src/util.cpp src/policy.cpp src/fk.cpp src/runlog.cpp)
target_include_directories(pm_policy PUBLIC include ${EIGEN3_INCLUDE_DIR} ${ORT_INCLUDE_DIR})
target_link_libraries(pm_policy PUBLIC ${ORT_LIB} Threads::Threads)

# ---- robot-facing core + the single executable (service | runner | guardian | release | selfcheck)
add_library(pm_robot STATIC src/guard.cpp src/link.cpp src/exit_policy.cpp src/handback.cpp src/session.cpp src/tts.cpp)
target_include_directories(pm_robot PUBLIC include ${SDK2_INCLUDE_DIR} ${DDSCXX_INCLUDE_DIR})
foreach(d ${DDSC_INCLUDE_DIR} ${ICEORYX_INCLUDE_DIR})
  if(d)
    target_include_directories(pm_robot PUBLIC ${d})
  endif()
endforeach()
target_link_libraries(pm_robot PUBLIC pm_policy ${SDK2_LIB} ${DDSCXX_LIB} ${DDSC_LIB} rt Threads::Threads)

add_executable(partymimic src/main.cpp src/runner.cpp src/guardian.cpp src/service.cpp src/selfcheck.cpp src/logtool.cpp)
target_link_libraries(partymimic PRIVATE pm_robot)

# development-only checks (not part of the robot package)
if(EXISTS ${CMAKE_SOURCE_DIR}/dev/CMakeLists.txt)
  add_subdirectory(dev)
endif()
